// T5 - Lacos de repeticao: while, do-while, for

int i;
int x;
int soma;

// while: contar de 0 a 4
i = 0;
while (i < 5) {
    write(i);
    i = i + 1;
}

// do-while: dobra x at ultrapassar 50
x = 1;
do {
    x = x * 2;
} while (x < 50);
write(x);           // esperado: 64

// for: soma 1..10
soma = 0;
for (i = 1; i <= 10; i = i + 1) {
    soma = soma + i;
}
write(soma);        // esperado: 55

// for aninhado: tabuada de 1 a 3
int j;
for (i = 1; i <= 3; i = i + 1) {
    for (j = 1; j <= 3; j = j + 1) {
        write(i * j);
    }
}
